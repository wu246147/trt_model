cmake_minimum_required(VERSION 3.20)
project(trt_model VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
if(MSVC)
    add_compile_options(/W3 /utf-8)
endif()
add_compile_definitions(NOMINMAX)
set(TRT_DIR "E:/env/dll/tensorRT/TensorRT-8.6.1.6" CACHE PATH "TensorRT 8.6.1.6 目录")
set(CUDAToolkit_ROOT "C:/Program Files/NVIDIA GPU Computing Toolkit/CUDA/v11.8" CACHE PATH "CUDA 目录")
set(OpenCV_DIR "D:/opencv4.5.5/opencv/build/x64/vc15/lib" CACHE PATH "OpenCV CMake 目录")
set(TRT_MODEL_QT_ROOT "D:/Qt/5.15.2/msvc2019_64" CACHE PATH "Qt MSVC x64 目录")
option(TRT_MODEL_BUILD_GUI "构建 Qt Designer 测试界面" ON)
option(TRT_MODEL_BUILD_TESTS "构建验证程序" ON)
find_package(CUDAToolkit REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs)
find_package(Threads REQUIRED)
find_path(TRT_INCLUDE_DIR NvInfer.h HINTS "${TRT_DIR}/include" NO_DEFAULT_PATH REQUIRED)
foreach(part nvinfer nvonnxparser nvinfer_plugin)
    find_library(TRT_${part}_LIBRARY NAMES ${part} HINTS "${TRT_DIR}/lib" NO_DEFAULT_PATH REQUIRED)
endforeach()
file(READ "${TRT_INCLUDE_DIR}/NvInferVersion.h" TRT_VERSION_HEADER)
foreach(part MAJOR MINOR PATCH BUILD)
    string(REGEX MATCH "#define NV_TENSORRT_${part} +([0-9]+)" _ "${TRT_VERSION_HEADER}")
    set(TRT_VERSION_${part} "${CMAKE_MATCH_1}")
endforeach()
if(NOT "${TRT_VERSION_MAJOR}.${TRT_VERSION_MINOR}.${TRT_VERSION_PATCH}.${TRT_VERSION_BUILD}" STREQUAL "8.6.1.6")
    message(FATAL_ERROR "本工程要求 TensorRT 8.6.1.6，请检查 TRT_DIR")
endif()

# 后端只在 DLL 内使用；调用者头文件不传播 CUDA、TensorRT 或解析器头文件。
add_library(trt_backend SHARED src/trt_engine.cpp src/trt_engine.h src/anomaly_onnx_prepare.cpp src/anomaly_onnx_prepare.h)
set_target_properties(trt_backend PROPERTIES POSITION_INDEPENDENT_CODE ON WINDOWS_EXPORT_ALL_SYMBOLS ON)
target_include_directories(trt_backend PUBLIC src PRIVATE "${TRT_INCLUDE_DIR}")
target_link_libraries(trt_backend PUBLIC CUDA::cudart "${TRT_nvinfer_LIBRARY}" "${TRT_nvonnxparser_LIBRARY}" "${TRT_nvinfer_plugin_LIBRARY}" Threads::Threads)
install(TARGETS trt_backend RUNTIME DESTINATION bin LIBRARY DESTINATION bin ARCHIVE DESTINATION lib)

foreach(version 5 11 26)
    if(version EQUAL 5)
        set(prefix yolov5)
    else()
        set(prefix yolo${version})
    endif()
    foreach(task det seg)
        if(task STREQUAL det)
            set(taskName Det)
        else()
            set(taskName Seg)
        endif()
        set(target RaivasDefect${taskName}${version})
        add_library(${target} SHARED src/${prefix}_${task}.cpp src/yolo_runner.cpp src/yolo_runner.h include/${prefix}_${task}.h)
        target_compile_definitions(${target} PRIVATE TRT_MODEL_EXPORTS)
        target_include_directories(${target} PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include> ${OpenCV_INCLUDE_DIRS})
        target_link_libraries(${target} PRIVATE trt_backend ${OpenCV_LIBS})
        list(APPEND YOLO_LIBRARIES ${target})
        install(TARGETS ${target} RUNTIME DESTINATION bin LIBRARY DESTINATION bin ARCHIVE DESTINATION lib)
    endforeach()
endforeach()
add_library(RaivasAnomalyTRT SHARED src/anomaly_trt.cpp src/anomaly_preprocess.cpp src/anomaly_preprocess.h include/anomaly_trt.h)
target_compile_definitions(RaivasAnomalyTRT PRIVATE ANOMALY_TRT_EXPORTS)
target_include_directories(RaivasAnomalyTRT PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include> ${OpenCV_INCLUDE_DIRS})
target_link_libraries(RaivasAnomalyTRT PRIVATE trt_backend ${OpenCV_LIBS})
install(TARGETS RaivasAnomalyTRT RUNTIME DESTINATION bin LIBRARY DESTINATION bin ARCHIVE DESTINATION lib)

add_executable(trt_engine_tool tools/engine_tool.cpp)
target_link_libraries(trt_engine_tool PRIVATE trt_backend)
install(TARGETS trt_engine_tool RUNTIME DESTINATION bin)

    add_executable(defect_alg_trt WIN32 gui/main.cpp gui/mainwindow.cpp gui/mainwindow.h gui/mainwindow.ui gui/imageview.cpp gui/imageview.h gui/guimodel.cpp gui/guimodel.h)
    set_target_properties(defect_alg_trt PROPERTIES AUTOMOC ON AUTOUIC ON)

    install(TARGETS defect_alg_trt RUNTIME DESTINATION bin)
